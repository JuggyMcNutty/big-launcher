cmake_minimum_required(VERSION 3.18)
project("foyer"
  LANGUAGES C CXX
  VERSION 3.0.0
)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EXECUTABLE_TITLE "foyer")
add_compile_definitions("$<$<CONFIG:DEBUG>:DEBUG>")

# v2 options
option(FOYER_CEC "Build with HDMI-CEC support (libcec)" OFF)
option(FOYER_EVDEV "Build with global evdev input (Home/Vol)" ON)
option(FOYER_WERROR "Treat warnings as errors (CI builds)" OFF)

# Warnings for Foyer's own targets only (not bundled SDL subprojects).
if (FOYER_WERROR)
  set(FOYER_WARNINGS -Wall -Wextra -Wpedantic -Werror)
else()
  set(FOYER_WARNINGS -Wall -Wextra)
endif ()

include(GNUInstallDirs)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
find_package(PkgConfig MODULE REQUIRED)
option(FOYER_BUNDLE_SDL "Build SDL3, SDL3_ttf and SDL3_image statically (portable glibc builds; distros without SDL3)" OFF)
if (FOYER_BUNDLE_SDL)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDLTTF_VENDORED OFF CACHE BOOL "" FORCE)       # system freetype + harfbuzz (stable sonames)
  set(SDLTTF_HARFBUZZ ON CACHE BOOL "" FORCE)
  set(SDLTTF_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_VENDORED OFF CACHE BOOL "" FORCE)     # system libpng/libjpeg/libwebp
  set(SDLIMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_TESTS OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_AVIF OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_JXL OFF CACHE BOOL "" FORCE)
  set(SDLIMAGE_TIF OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.10/SDL3-3.4.10.tar.gz)
  FetchContent_Declare(SDL3_ttf URL https://github.com/libsdl-org/SDL_ttf/releases/download/release-3.2.2/SDL3_ttf-3.2.2.tar.gz)
  FetchContent_Declare(SDL3_image URL https://github.com/libsdl-org/SDL_image/releases/download/release-3.4.8/SDL3_image-3.4.8.tar.gz)
  FetchContent_MakeAvailable(SDL3 SDL3_ttf SDL3_image)
  add_library(foyer_sdl INTERFACE)
  target_link_libraries(foyer_sdl INTERFACE SDL3_image::SDL3_image-static SDL3_ttf::SDL3_ttf-static SDL3::SDL3-static)
  add_library(PkgConfig::SDL3 ALIAS foyer_sdl)
  add_library(foyer_sdl_none INTERFACE)
  add_library(PkgConfig::SDL3_IMAGE ALIAS foyer_sdl_none)
  add_library(PkgConfig::SDL3_TTF ALIAS foyer_sdl_none)
else ()
  pkg_check_modules(SDL3 REQUIRED IMPORTED_TARGET sdl3)
  pkg_check_modules(SDL3_IMAGE REQUIRED IMPORTED_TARGET sdl3-image)
  pkg_check_modules(SDL3_TTF REQUIRED IMPORTED_TARGET sdl3-ttf)
endif ()
if (FOYER_BUNDLE_SDL)
  # Portable build: only depend on sonames that are stable across glibc
  # distros (libc, libstdc++, freetype, harfbuzz). fmt + spdlog header-only,
  # libxml2 static.
  # libxml2 without ICU/lzma/zlib, static (only the parser/tree API is used)
  set(LIBXML2_WITH_ICU OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_LZMA OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_ZLIB OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_ICONV OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_PYTHON OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_TESTS OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_HTTP OFF CACHE BOOL "" FORCE)
  set(LIBXML2_WITH_MODULES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(libxml2 URL https://download.gnome.org/sources/libxml2/2.13/libxml2-2.13.9.tar.xz)
  FetchContent_MakeAvailable(libxml2)
  add_library(PkgConfig::LIBXML2 ALIAS LibXml2)
  find_path(FMT_INCLUDE fmt/format.h REQUIRED)
  find_path(SPDLOG_INCLUDE spdlog/spdlog.h REQUIRED)
  add_library(foyer_fmt INTERFACE)
  target_include_directories(foyer_fmt INTERFACE ${FMT_INCLUDE})
  target_compile_definitions(foyer_fmt INTERFACE FMT_HEADER_ONLY)
  add_library(PkgConfig::FMT ALIAS foyer_fmt)
  add_library(foyer_spdlog INTERFACE)
  target_include_directories(foyer_spdlog INTERFACE ${SPDLOG_INCLUDE})
  target_compile_definitions(foyer_spdlog INTERFACE SPDLOG_HEADER_ONLY SPDLOG_FMT_EXTERNAL FMT_HEADER_ONLY)
  add_library(PkgConfig::SPDLOG ALIAS foyer_spdlog)
else ()
  pkg_check_modules(LIBXML2 REQUIRED IMPORTED_TARGET libxml-2.0)
  pkg_check_modules(FMT REQUIRED IMPORTED_TARGET fmt)
  pkg_check_modules(SPDLOG REQUIRED IMPORTED_TARGET spdlog)
endif ()

if (FOYER_EVDEV)
  pkg_check_modules(LIBEVDEV QUIET IMPORTED_TARGET libevdev)
  if (LIBEVDEV_FOUND)
    add_compile_definitions(FOYER_EVDEV)
  else()
    message(WARNING "libevdev not found; FOYER_EVDEV disabled")
  endif ()
endif ()

if (FOYER_CEC)
  pkg_check_modules(LIBCEC REQUIRED IMPORTED_TARGET libcec)
  add_compile_definitions(FOYER_CEC)
endif ()

configure_file(${PROJECT_SOURCE_DIR}/config/config.h.in ${PROJECT_BINARY_DIR}/lconfig.h)

add_subdirectory(src)

# Tests
option(FOYER_TESTS "Build unit tests" ON)
if (FOYER_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif ()
